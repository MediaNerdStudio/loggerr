cmake_minimum_required(VERSION 3.24)
project(LoggerrPortAudio LANGUAGES C CXX)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5 CACHE STRING "" FORCE)
include(FetchContent)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(PA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(PA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(PA_BUILD_SHARED ON CACHE BOOL "" FORCE)
set(PA_BUILD_STATIC OFF CACHE BOOL "" FORCE)
set(PA_USE_WMME ON CACHE BOOL "" FORCE)
set(PA_USE_WASAPI ON CACHE BOOL "" FORCE)
set(PA_USE_WDMKS ON CACHE BOOL "" FORCE)
option(LOGGERR_ENABLE_ASIO "Build PortAudio with a locally supplied Steinberg ASIO SDK" OFF)
if(LOGGERR_ENABLE_ASIO)
  if(NOT ASIO_SDK_PATH)
    message(FATAL_ERROR "Set ASIO_SDK_PATH to an approved local Steinberg ASIO SDK checkout")
  endif()
  set(PA_USE_ASIO ON CACHE BOOL "" FORCE)
  set(ASIOSDK_ROOT_DIR ${ASIO_SDK_PATH} CACHE PATH "" FORCE)
else()
  set(PA_USE_ASIO OFF CACHE BOOL "" FORCE)
endif()
FetchContent_Declare(portaudio GIT_REPOSITORY https://github.com/PortAudio/portaudio.git GIT_TAG v19.7.0 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(portaudio)
